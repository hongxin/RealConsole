```rust
use chrono::{DateTime, Utc};
use serde::{Deserialize, Serialize};
use std::fs;
use std::path::PathBuf;

/// ✨ v1.22.0: 保存的任务数据
#[derive(Debug, Clone, Serialize, Deserialize)]
pub struct SavedTask {
    /// 任务 ID (时间戳格式: YYYYMMDD_HHMMSS)
    pub id: String,

    /// 任务名称（用户指定，可选）
    pub name: Option<String>,

    /// 创建时间
    pub created_at: DateTime<Utc>,

    /// 任务目标描述
    pub goal: String,

    /// 任务计划
    pub plan: ExecutionPlan,

    /// 执行结果（如果已执行）
    pub result: Option<ExecutionResult>,
}

impl SavedTask {
    /// 创建新的保存任务
    pub fn new(goal: String, plan: ExecutionPlan, result: Option<ExecutionResult>) -> Self {
        let now = Utc::now();
        let id = now.format("%Y%m%d_%H%M%S").to_string();

        Self {
            id,
            name: None,
            created_at: now,
            goal,
            plan,
            result,
        }
    }

    /// 设置任务名称
    pub fn with_name(mut self, name: String) -> Self {
        self.name = Some(name);
        self
    }

    /// 保存到文件
    pub fn save_to_file(&self, storage_path: &PathBuf) -> Result<(), String> {
        // 确保目录存在
        fs::create_dir_all(storage_path)
            .map_err(|e| format!("创建目录失败: {}", e))?;

        // 文件路径
        let file_path = storage_path.join(format!("{}.json", self.id));

        // 序列化为 JSON
        let json = serde_json::to_string_pretty(self)
            .map_err(|e| format!("序列化失败: {}", e))?;

        // 写入文件
        fs::write(&file_path, json)
            .map_err(|e| format!("写入文件失败: {}", e))?;

        Ok(())
    }

    /// 从文件加载
    pub fn load_from_file(file_path: &PathBuf) -> Result<Self, String> {
        let json = fs::read_to_string(file_path)
            .map_err(|e| format!("读取文件失败: {}", e))?;

        let task: SavedTask = serde_json::from_str(&json)
            .map_err(|e| format!("解析 JSON 失败: {}", e))?;

        Ok(task)
    }

    /// 列出所有保存的任务
    pub fn list_all(storage_path: &PathBuf) -> Result<Vec<SavedTask>, String> {
        if !storage_path.exists() {
            return Ok(Vec::new());
        }

        let mut tasks = Vec::new();

        for entry in fs::read_dir(storage_path)
            .map_err(|e| format!("读取目录失败: {}", e))?
        {
            let entry = entry.map_err(|e| format!("读取条目失败: {}", e))?;
            let path = entry.path();

            if path.extension().and_then(|s| s.to_str()) == Some("json") {
                if let Ok(task) = Self::load_from_file(&path) {
                    tasks.push(task);
                }
            }
        }

        // 按创建时间倒序排序
        tasks.sort_by(|a, b| b.created_at.cmp(&a.created_at));

        Ok(tasks)
    }
}
```

#### 1.2 TaskManager 增强

**文件**: `src/commands/task_cmd.rs`

```rust
impl TaskManager {
    /// ✨ v1.22.0: 获取存储路径
    fn get_storage_path() -> PathBuf {
        // 使用 ~/.realconsole/tasks
        let home = dirs::home_dir().unwrap_or_else(|| PathBuf::from("."));
        home.join(".realconsole").join("tasks")
    }

    /// ✨ v1.22.0: 保存当前任务到文件
    pub fn save_current_task(&self, name: Option<String>) -> Result<String, String> {
        let plan = self.current_plan.as_ref()
            .ok_or_else(|| "没有当前任务计划".to_string())?;

        let goal = plan.goal.clone();
        let result = self.last_result.clone();

        let mut saved_task = SavedTask::new(goal, plan.clone(), result);
        if let Some(n) = name {
            saved_task = saved_task.with_name(n);
        }

        let storage_path = Self::get_storage_path();
        saved_task.save_to_file(&storage_path)?;

        Ok(saved_task.id)
    }

    /// ✨ v1.22.0: 列出所有保存的任务
    pub fn list_saved_tasks() -> Result<Vec<SavedTask>, String> {
        let storage_path = Self::get_storage_path();
        SavedTask::list_all(&storage_path)
    }

    /// ✨ v1.22.0: 加载指定任务
    pub fn load_task(&mut self, task_id: &str) -> Result<(), String> {
        let storage_path = Self::get_storage_path();
        let file_path = storage_path.join(format!("{}.json", task_id));

        let saved_task = SavedTask::load_from_file(&file_path)?;

        // 加载计划和结果
        self.current_plan = Some(saved_task.plan);
        self.last_result = saved_task.result;

        Ok(())
    }
}
```

##### `/task_save [name]` 命令

```rust
// 在 register_task_commands 中注册
{
    let manager = Arc::clone(&task_manager);

    registry.register(Command::from_fn(
        "task_save",
        "保存当前任务",
        move |arg: &str| {
            let manager = Arc::clone(&manager);

            tokio::task::block_in_place(|| {
                tokio::runtime::Handle::current()
                    .block_on(async {
                        let mgr = manager.read().await;

                        let name = if arg.trim().is_empty() {
                            None
                        } else {
                            Some(arg.trim().to_string())
                        };

                        match mgr.save_current_task(name) {
                            Ok(task_id) => {
                                format!(
                                    "{} {}\n{} {}",
                                    "✓".green(),
                                    "任务已保存".bold(),
                                    "任务 ID:".dimmed(),
                                    task_id.yellow()
                                )
                            }
                            Err(e) => {
                                format!("{} {}", "[ERROR]".red(), e)
                            }
                        }
                    })
            })
        },
    ));
}
```

##### `/task_list` 命令

```rust
{
    registry.register(Command::from_fn(
        "task_list",
        "列出所有保存的任务",
        move |_arg: &str| {
            match TaskManager::list_saved_tasks() {
                Ok(tasks) => {
                    if tasks.is_empty() {
                        format!("{}", "没有保存的任务".dimmed())
                    } else {
                        let mut output = String::new();
                        output.push_str(&format!("\n{}\n\n", "保存的任务".bold()));

                        for (idx, task) in tasks.iter().enumerate() {
                            let status_icon = if task.result.is_some() {
                                "✓".green()
                            } else {
                                "○".dimmed()
                            };

                            let name_display = task.name.as_ref()
                                .map(|n| format!("{} ", n.bold()))
                                .unwrap_or_default();

                            output.push_str(&format!(
                                "  {} {} {}{}\n",
                                (idx + 1).to_string().dimmed(),
                                status_icon,
                                name_display,
                                task.id.yellow()
                            ));

                            output.push_str(&format!(
                                "      {} {}\n",
                                "目标:".dimmed(),
                                task.goal
                            ));

                            output.push_str(&format!(
                                "      {} {}\n",
                                "创建:".dimmed(),
                                task.created_at.format("%Y-%m-%d %H:%M:%S").to_string().dimmed()
                            ));

                            if task.result.is_some() {
                                output.push_str(&format!(
                                    "      {}\n",
                                    "已执行".green().dimmed()
                                ));
                            }

                            output.push('\n');
                        }

                        output.push_str(&format!(
                            "\n{} {}\n",
                            "总计:".dimmed(),
                            format!("{} 个任务", tasks.len()).bold()
                        ));

                        output.push_str(&format!(
                            "{}\n",
                            "使用 /task_load <id> 加载任务".dimmed()
                        ));

                        output
                    }
                }
                Err(e) => {
                    format!("{} {}", "[ERROR]".red(), e)
                }
            }
        },
    ));
}
```

##### `/task_load <id>` 命令

```rust
{
    let manager = Arc::clone(&task_manager);

    registry.register(Command::from_fn(
        "task_load",
        "加载保存的任务",
        move |arg: &str| {
            let manager = Arc::clone(&manager);

            tokio::task::block_in_place(|| {
                tokio::runtime::Handle::current()
                    .block_on(async {
                        let task_id = arg.trim();

                        if task_id.is_empty() {
                            return format!(
                                "{}\n使用方式: /task_load <id>",
                                "[ERROR] 请提供任务 ID".red()
                            );
                        }

                        let mut mgr = manager.write().await;

                        match mgr.load_task(task_id) {
                            Ok(_) => {
                                let plan = mgr.get_current_plan().unwrap();

                                    "{} {}\n\n{} {}\n{} {} 个任务\n\n{}",
                                    "✓".green(),

                                    "目标:".dimmed(),
                                    plan.goal.bold(),
                                    "任务数:".dimmed(),
                                    plan.stages.iter().map(|s| s.tasks.len()).sum::<usize>(),
                                    "使用 /execute 执行任务".dimmed()
                                )
                            }
                            Err(e) => {
                                format!("{} {}", "[ERROR]".red(), e)
                            }
                        }
                    })
            })
        },
    ));
}
```

```rust
use regex::Regex;

/// 高亮字符串中的数字
fn highlight_numbers(text: &str) -> String {
    let re = Regex::new(r"\b\d+(?:\.\d+)?\b").unwrap();

    re.replace_all(text, |caps: &regex::Captures| {
        format!("{}", caps[0].yellow().bold())
    }).to_string()
}

// 在 task_execution_result 中使用
if config.task.display.highlight_numbers {
    println!("    {}", highlight_numbers(line).dimmed());
} else {
    println!("    {}", line.dimmed());
}
```
