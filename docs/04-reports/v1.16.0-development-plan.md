pub struct ConfigWizard {

    recommender: ConfigRecommender,
    validator: ConfigValidator,

pub struct EnvironmentDetector {
    // 检测系统环境
    pub fn detect_os(&self) -> OsInfo;
    pub fn detect_shell(&self) -> ShellInfo;
    pub fn detect_tools(&self) -> Vec<InstalledTool>;
    pub fn detect_user_profile(&self) -> UserProfile;
}

pub struct ConfigRecommender {
    // AI驱动的配置推荐
    pub async fn recommend_llm(&self, env: &EnvInfo) -> LlmRecommendation;
    pub async fn recommend_mode(&self, profile: &UserProfile) -> ModeRecommendation;

pub struct ConfigValidator {
    // 配置验证
    pub async fn validate_llm(&self, config: &LlmConfig) -> Result<()>;
    pub async fn validate_api_key(&self, key: &str) -> Result<()>;
}

// 配置流程
pub async fn run_wizard() -> Result<Config> {
    let wizard = ConfigWizard::new();

    // 1. 环境检测
    let env = wizard.detector.detect_all()?;

    // 2. 推荐配置
    let recommendations = wizard.recommender.recommend_all(&env).await?;

    // 3. 交互式配置
    let config = wizard.interactive_config(recommendations).await?;

    // 4. 验证配置
    wizard.validator.validate_all(&config).await?;

    // 5. 保存配置
    config.save()?;

    Ok(config)
}

**先进性设计**：

1. **AI驱动的错误分析**
   ```rust
   pub struct ErrorAnalyzer {
       pattern_matcher: PatternMatcher,    // 错误模式匹配
       llm_analyzer: LlmAnalyzer,          // LLM深度分析
       context_aware: ContextAware,         // 上下文感知
       predictor: NextStepPredictor,        // 下一步预测
   }

   impl ErrorAnalyzer {
       pub async fn analyze(&self, error: &CommandError) -> ErrorAnalysis {
           // 1. 快速模式匹配（本地，<10ms）
           if let Some(quick_fix) = self.pattern_matcher.match_error(error) {
               return quick_fix;
           }

           // 2. 上下文增强（项目类型、历史命令）
           let context = self.context_aware.gather_context();

           // 3. LLM深度分析（在线，<500ms）
           let analysis = self.llm_analyzer.analyze(error, &context).await?;

           // 4. 预测下一步
           let next_steps = self.predictor.predict(&analysis);

           ErrorAnalysis {
               root_cause: analysis.cause,
               suggestions: analysis.suggestions,
               next_steps,
               risk_level: self.assess_risk(error),
           }
       }
   }
   ```

2. **多维度建议排序**
   - **相关性**（0.0-1.0）：与当前错误的相关度
   - **可行性**（0.0-1.0）：成功概率
   - **安全性**（0.0-1.0）：操作风险评估
   - **学习价值**（0.0-1.0）：是否帮助用户学习

3. **风险分级与提示**
   | 风险等级 | 符号 | 颜色 | 行为 |
   |---------|------|------|------|
   | 安全 | ✅ | 绿色 | 直接执行 |
   | 低风险 | ⚠️ | 黄色 | 提示确认 |
   | 中风险 | 🔶 | 橙色 | 详细说明 |
   | 高风险 | 🔴 | 红色 | 强制二次确认 |

**技术实现**：
```rust
// src/error/enhanced_error.rs

pub struct EnhancedError {
    pub original_error: CommandError,
    pub analysis: ErrorAnalysis,
    pub suggestions: Vec<Suggestion>,
    pub next_steps: Vec<NextStep>,
    pub risk_assessment: RiskAssessment,
}

pub struct ErrorAnalysis {
    pub root_cause: String,
    pub confidence: f64,           // 分析置信度
    pub context: ErrorContext,     // 错误上下文
}

pub struct Suggestion {
    pub command: String,
    pub description: String,
    pub relevance: f64,            // 相关性评分
    pub feasibility: f64,          // 可行性评分
    pub safety: f64,               // 安全性评分
    pub learning_value: f64,       // 学习价值评分
}

impl Suggestion {
    // 综合评分
    pub fn score(&self) -> f64 {
        self.relevance * 0.4 +
        self.feasibility * 0.3 +
        self.safety * 0.2 +
        self.learning_value * 0.1
    }
}

pub struct NextStep {
    pub action: String,
    pub probability: f64,          // 用户可能执行的概率
    pub benefit: String,           // 执行的好处
}

// 预测引擎
pub struct NextStepPredictor {
    history_analyzer: HistoryAnalyzer,
    pattern_learner: PatternLearner,
}

impl NextStepPredictor {
    pub fn predict(&self, analysis: &ErrorAnalysis) -> Vec<NextStep> {
        // 基于历史行为预测
        let history_patterns = self.history_analyzer.find_patterns();

        // 基于错误类型预测
        let typical_next = self.get_typical_next_steps(&analysis.root_cause);

        // 融合预测结果
        self.merge_predictions(history_patterns, typical_next)
    }
}
```

#### 3. 修复测试并发问题（4天）

**问题分析**：

当前有 7-9 个 race condition 导致并行测试失败：

```bash
# 当前状态
$ cargo test
running 1136 tests
test result: FAILED. 1127 passed; 9 failed; 0 ignored

# 问题原因
1. 共享状态竞争（Agent内部的RwLock）
2. 文件系统竞争（Memory、Bagua同时写入）
3. 时间相关的测试（snapshot_interval竞争）
```

**解决方案**：

1. **隔离测试环境**
   ```rust
   // tests/test_helper.rs

   pub struct TestEnv {
       temp_dir: TempDir,
       config_file: PathBuf,
       memory_dir: PathBuf,
   }

   impl TestEnv {
       pub fn new(test_name: &str) -> Self {
           // 每个测试独立的临时目录
           let temp_dir = TempDir::new()
               .unwrap()
               .into_path()
               .join(test_name);

           fs::create_dir_all(&temp_dir).unwrap();

           Self {
               temp_dir: temp_dir.clone(),
               config_file: temp_dir.join("config.yaml"),
               memory_dir: temp_dir.join("memory"),
           }
       }

       pub fn cleanup(self) {
           // 测试结束后自动清理
           let _ = fs::remove_dir_all(&self.temp_dir);
       }
   }

   // 使用
   #[tokio::test]
   async fn test_something() {
       let env = TestEnv::new("test_something");
       // ... 测试代码
       env.cleanup();
   }
   ```

2. **锁优化**
   ```rust
   // 当前问题
   let state = self.state.write().await;  // 长时间持有写锁
   heavy_computation();
   drop(state);

   // 优化后
   let data = {
       let state = self.state.read().await;  // 短暂持有读锁
       state.clone()  // 克隆需要的数据
   };
   heavy_computation();  // 不持有锁
   ```

3. **测试并发控制**
   ```rust
   // Cargo.toml
   [dev-dependencies]
   serial_test = "3.0"

   // 冲突的测试强制串行
   use serial_test::serial;

   #[tokio::test]
   #[serial(memory_tests)]  // 同组测试串行执行
   async fn test_memory_write() {
       // ...
   }

   #[tokio::test]
   #[serial(memory_tests)]
   async fn test_memory_read() {
       // ...
   }
   ```

4. **性能优化（顺带）**
   - 减少不必要的 async/await
   - 使用 `tokio::test(flavor = "multi_thread")` 合理配置
   - 长时间测试标记为 `#[ignore]`

**目标**：
- ✅ 并行测试 100% 通过（`cargo test` 无 `--test-threads=1`）
- ✅ 测试时间从 226s 降低到 <100s
- ✅ 零 flaky tests（不稳定的测试）

### P1 - 重要但可延后

#### 4. Memory系统重构决策（3天）

**当前问题**：
- Memory 与 Trace 功能重叠（都记录历史）
- Memory 与 Bagua 功能重叠（都存储数据）
- 查询接口不统一，用户困惑

**方案对比**：

| 方案 | 优点 | 缺点 | 工作量 | 推荐 |
|------|------|------|--------|------|
| **A. 合并到Trace** | 统一查询接口 | Trace变复杂 | 2天 | ⭐⭐⭐ |
| **B. 合并到Bagua** | 统一存储系统 | 需要重新设计 | 4天 | ⭐⭐ |
| **C. 保留并简化** | 改动最小 | 仍有重叠 | 1天 | ⭐ |
| **D. 完全移除** | 最简单 | 丢失部分功能 | 0.5天 | ⭐ |

**推荐方案A详细设计**：

```rust
// src/tracer/unified_tracer.rs

// 扩展 Dimension 枚举
pub enum Dimension {
    History,       // 命令历史（已有）
    Log,           // 执行日志（已有）
    LlmLog,        // LLM日志（已有）
    Context,       // 对话上下文（已有）
    Memory,        // 记忆系统（新增） ⬅️
    Custom(String),// 自定义维度
}

// 扩展 TraceEntry
pub struct TraceEntry {
    pub id: String,
    pub dimension: Dimension,
    pub entry_type: EntryType,
    pub content: String,
    pub timestamp: DateTime<Utc>,
    pub metadata: HashMap<String, String>,

    // 新增：Memory相关字段
    pub importance: Option<Importance>,  // 重要性级别
    pub tags: Vec<String>,               // 标签
    pub embedding: Option<Vec<f32>>,     // 语义向量（未来）
}

// Memory查询API保持不变，但底层使用Trace
pub struct MemoryManager {
    tracer: Arc<RwLock<UnifiedTracer>>,
}

impl MemoryManager {
    // 保持原有API，避免破坏性变更
    pub async fn store(&self, entry: MemoryEntry) -> Result<()> {
        let trace_entry = self.convert_to_trace_entry(entry);
        self.tracer.write().await.add_entry(trace_entry)?;
        Ok(())
    }

    pub async fn search(&self, keyword: &str) -> Result<Vec<MemoryEntry>> {
        let entries = self.tracer.read().await
            .search(keyword)
            .filter(|e| e.dimension == Dimension::Memory)
            .collect();
        Ok(self.convert_from_trace_entries(entries))
    }

    // 新增：基于重要性查询
    pub async fn find_important(&self, min_level: Importance) -> Result<Vec<MemoryEntry>> {
        let entries = self.tracer.read().await
            .get_entries(Dimension::Memory, None)
            .filter(|e| e.importance.unwrap_or_default() >= min_level)
            .collect();
        Ok(self.convert_from_trace_entries(entries))
    }
}
```

2. Day 2: 实现 `MemoryManager` 适配层，保持API兼容
3. Day 3: 测试、迁移现有数据、文档更新
