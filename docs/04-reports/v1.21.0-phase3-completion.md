### 1. 配置结构定义

新增 3 个配置结构：

```rust
/// 任务系统配置
#[derive(Debug, Clone, Default, Serialize, Deserialize)]
pub struct TaskConfig {
    pub display: TaskDisplayConfig,
    pub execution: TaskExecutionConfig,
}

/// 任务显示配置
#[derive(Debug, Clone, Serialize, Deserialize)]
pub struct TaskDisplayConfig {
    #[serde(default = "default_true")]
    pub show_task_output: bool,       // 是否显示任务输出（默认 true）

    #[serde(default = "default_max_output_lines")]
    pub max_output_lines: usize,      // 最大输出行数（默认 50，0=无限制）

    #[serde(default = "default_true")]
    pub highlight_numbers: bool,      // 是否高亮数字（默认 true）

    #[serde(default = "default_true")]
    pub show_task_duration: bool,     // 是否显示执行时间（默认 true）
}

/// 任务执行配置
#[derive(Debug, Clone, Serialize, Deserialize)]
pub struct TaskExecutionConfig {
    #[serde(default = "default_true")]
    pub merge_stages: bool,           // 是否合并 Stage（默认 true）

    #[serde(default = "default_max_merged_tasks")]
    pub max_merged_tasks: usize,      // 最大合并任务数（默认 20）
}
```

**关键代码**:
```rust
// 使用配置控制显示
if config.task.display.show_task_output && !result.task_results.is_empty() {
    // 应用行数限制
    let max_lines = if config.task.display.max_output_lines == 0 {
        usize::MAX  // 0 表示不限制
    } else {
        config.task.display.max_output_lines
    };

    // 显示省略提示
    if lines.len() > max_lines {
        println!(
            "    {} (省略 {} 行，使用 /task_output {} 查看完整输出)",
            "...".dimmed(),
            lines.len() - max_lines,
            task_idx
        );
    }
}
```
