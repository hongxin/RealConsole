**目标**: 支持查看任务完整输出，突破 50 行限制

```rust
// src/agent.rs
pub async fn handle_system_command(&mut self, cmd: &str) -> AgentResult<()> {
    match cmd {
        s if s.starts_with("/task_output ") => {
            let id_str = s.trim_start_matches("/task_output ").trim();
            let task_id: usize = id_str.parse()
                .map_err(|_| AgentError::InvalidArgument(format!("无效的任务 ID: {}", id_str)))?;
            self.show_task_output(task_id).await?;
        }
        // ...
    }
}

async fn show_task_output(&self, task_id: usize) -> AgentResult<()> {
    // 1. 从 state.last_execution_result 中获取任务结果
    let Some(result) = self.state.last_execution_result.as_ref() else {
        return Err(AgentError::InvalidArgument("没有可用的任务执行结果".to_string()));
    };

    // 2. 查找指定 ID 的任务
    let Some(task_result) = result.task_results.get(task_id) else {
        return Err(AgentError::InvalidArgument(format!("任务 ID {} 不存在（有效范围: 0-{}）",
            task_id, result.task_results.len() - 1)));
    };

    // 3. 显示完整输出
    println!("\n{}", "=".repeat(80).dimmed());
    println!("{} {} (ID: {})",
        task_status_icon(&task_result.status),
        task_result.task.name.bold(),
        task_id.to_string().dimmed()
    );
    println!("{}", "=".repeat(80).dimmed());

    if !task_result.output.trim().is_empty() {
        println!("\n{}", task_result.output);  // 完整输出，无行数限制
    } else {
        println!("\n{}", "（无输出）".dimmed());
    }

    if let Some(error) = &task_result.error {
        println!("\n{} {}", "错误:".red().bold(), error.red());
    }

    println!("\n{}", "=".repeat(80).dimmed());

    Ok(())
}
```

```rust
// src/task/persistence.rs
use serde::{Deserialize, Serialize};
use std::path::PathBuf;

#[derive(Debug, Clone, Serialize, Deserialize)]
pub struct PersistedTask {
    pub id: String,  // UUID
    pub created_at: chrono::DateTime<chrono::Utc>,
    pub plan: TaskPlan,
    pub execution_state: Option<ExecutionState>,
}

#[derive(Debug, Clone, Serialize, Deserialize)]
pub struct ExecutionState {
    pub started_at: chrono::DateTime<chrono::Utc>,
    pub completed_stages: Vec<usize>,  // 已完成的 Stage ID
    pub current_stage: Option<usize>,
    pub status: ExecutionStatus,
}

#[derive(Debug, Clone, Serialize, Deserialize)]
pub enum ExecutionStatus {
    NotStarted,
    InProgress,
    Paused,
    Completed,
    Failed,
}

pub struct TaskPersistence {
    storage_path: PathBuf,
}

impl TaskPersistence {
    pub fn new(config: &Config) -> Self {
        let storage_path = config.get_home_dir().join("tasks");
        std::fs::create_dir_all(&storage_path).ok();
        Self { storage_path }
    }

    pub fn save(&self, task: &PersistedTask) -> Result<()> {
        let file_path = self.storage_path.join(format!("{}.json", task.id));
        let json = serde_json::to_string_pretty(task)?;
        std::fs::write(file_path, json)?;
        Ok(())
    }

    pub fn load(&self, id: &str) -> Result<PersistedTask> {
        let file_path = self.storage_path.join(format!("{}.json", id));
        let json = std::fs::read_to_string(file_path)?;
        let task = serde_json::from_str(&json)?;
        Ok(task)
    }

    pub fn list(&self) -> Result<Vec<PersistedTask>> {
        let mut tasks = Vec::new();
        for entry in std::fs::read_dir(&self.storage_path)? {
            let entry = entry?;
            if entry.path().extension() == Some(std::ffi::OsStr::new("json")) {
                if let Ok(task) = self.load(&entry.file_name().to_string_lossy().trim_end_matches(".json")) {
                    tasks.push(task);
                }
            }
        }
        tasks.sort_by(|a, b| b.created_at.cmp(&a.created_at));
        Ok(tasks)
    }
}
```
