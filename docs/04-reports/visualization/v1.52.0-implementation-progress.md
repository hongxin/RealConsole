```rust
/// v1.52.0: 提取并处理图像数据
fn extract_and_process_image_data(content: &str) -> (String, Option<visualization::ImageData>) {
    if let Some(start) = content.find("__IMAGE__") {
        let json_start = start + "__IMAGE__".len();
        if let Some(end) = content[json_start..].find("__DEBUG__") {
            let json_str = &content[json_start..json_start + end];

            // 解析 ImageData JSON
            if let Ok(tool_params) = serde_json::from_str::<serde_json::Value>(json_str) {
                if let Ok(image_data) = serde_json::from_value::<visualization::ImageData>(tool_params) {
                    // 验证数据
                    if let Err(e) = image_data.validate() {
                        log::error!("[v1.52.0] ImageData 验证失败: {}", e);
                        return (content.to_string(), None);
                    }

                    // 清理内容（移除 __IMAGE__ 标记）
                    let clean_content = format!(
                        "{}{}",
                        &content[..start].trim_end(),
                        &content[json_start + end..]
                    );

                    return (clean_content, Some(image_data));
                }
            }
        }
    }
    (content.to_string(), None)
}
```

```javascript
renderImage(msg) {
    console.log(`[v1.52.0] Rendering image for round: ${msg.round_id}`);
    const imageData = msg.image_data;

    // 保存图像数据到追踪器（用于 localStorage 持久化）
    this.imageDataByRound[msg.round_id] = imageData;

    // 找到 Round 卡片
    const roundElement = document.getElementById(`round-${msg.round_id}`);
    if (!roundElement) {
        console.error(`[v1.52.0 ERROR] Round not found: ${msg.round_id}`);
        return;
    }

    // 找到输出区域
    const outputContent = roundElement.querySelector('.output-content');
    if (!outputContent) {
        console.error(`[v1.52.0 ERROR] Output content not found for round: ${msg.round_id}`);
        return;
    }

    // 创建图像容器
    const imageCard = document.createElement('div');
    imageCard.className = 'image-card';

    // 创建图像元素
    const img = document.createElement('img');
    img.className = 'display-image';
    img.alt = imageData.alt_text || '图像';

    // 设置图像源
    if (imageData.image_type === 'base64') {
        img.src = `data:${imageData.mime_type};base64,${imageData.data}`;
    } else if (imageData.image_type === 'url') {
        img.src = imageData.data;
    }

    // 加载状态处理
    img.onload = () => {
        imageCard.classList.add('loaded');
    };

    img.onerror = () => {
        imageCard.classList.add('error');
        const errorMsg = document.createElement('div');
        errorMsg.className = 'image-error';
        errorMsg.textContent = `图像加载失败: ${imageData.filename || ''}`;
        imageCard.appendChild(errorMsg);
    };

    imageCard.appendChild(img);

    // 添加文件名说明（如果有）
    if (imageData.filename) {
        const caption = document.createElement('div');
        caption.className = 'image-caption';
        caption.textContent = imageData.filename;
        if (imageData.size_bytes) {
            const sizeMB = (imageData.size_bytes / (1024 * 1024)).toFixed(2);
            caption.textContent += ` (${sizeMB} MB)`;
        }
        imageCard.appendChild(caption);
    }

    outputContent.appendChild(imageCard);
    this.scrollToBottom();
}
```
